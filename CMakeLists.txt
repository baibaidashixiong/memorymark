cmake_minimum_required(VERSION 3.13)
project(MemoryMark VERSION 1.3.1 LANGUAGES C)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(MM_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
if(MINGW)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
add_library(mm STATIC src/cli.c src/timer.c src/platform.c src/memory.c
  src/bench_seq.c src/bench_random.c src/bench_latency.c src/bench_parallel.c src/output.c src/score.c)
target_include_directories(mm PUBLIC include)
target_compile_definitions(mm PUBLIC MM_VERSION="${PROJECT_VERSION}" MM_BUILD="${CMAKE_BUILD_TYPE}")
target_link_libraries(mm PUBLIC Threads::Threads)
if(NOT WIN32)
  target_compile_definitions(mm PUBLIC _GNU_SOURCE)
  target_link_libraries(mm PUBLIC m)
endif()
if(MSVC)
  target_compile_options(mm PRIVATE /W4)
elseif(MINGW)
  target_compile_options(mm PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -static -static-libgcc)
  target_link_options(mm PUBLIC -static -static-libgcc)
else()
  target_compile_options(mm PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(MM_SANITIZE)
    target_compile_options(mm PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mm PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(memorymark src/main.c)
target_link_libraries(memorymark PRIVATE mm)
enable_testing()
add_executable(mm_unit tests/unit.c)
target_link_libraries(mm_unit PRIVATE mm)
add_test(NAME unit COMMAND mm_unit)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:memorymark>)
  add_test(NAME score-smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/score_smoke.py $<TARGET_FILE:memorymark>)
  set_tests_properties(score-smoke PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
endif()
